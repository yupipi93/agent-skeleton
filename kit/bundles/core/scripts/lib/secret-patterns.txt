# Secret patterns for kit/bundles/core/scripts/text-guard.sh and lib/secretscan.py.
#
# Format: <label> <regex>
#   label = first word of the line; regex = the rest of the line.
# Regexes are POSIX extended (grep -E on macOS and Linux): use [[:space:]] instead of
# \s, no lookarounds, no \d. They are case-sensitive on purpose: the gate is blunt, and
# a case-insensitive version blocks far too much prose.
# The em dash style check is not here: text-guard.sh adds it with --em-dash.

atlassian-token         (ATATT|ATCTT)[A-Za-z0-9_=-]{8,}
aws-access-key          AKIA[0-9A-Z]{12,}
aws-secret-field        aws_secret_access_key|aws_session_token
google-api-key          AIza[0-9A-Za-z_-]{30,}
gcp-service-account     "type"[[:space:]]*:[[:space:]]*"service_account"
api-key-sk              sk-(ant-)?[A-Za-z0-9_-]{16,}
slack-token             xox[baprs]-[A-Za-z0-9-]{8,}
github-token            gh[pousr]_[A-Za-z0-9]{20,}
jwt                     eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+
private-key             BEGIN [A-Z ]*PRIVATE KEY
bearer-token            [Bb]earer[[:space:]]+[A-Za-z0-9._-]{10,}
credential-assignment   (password|passwd|secret|client_secret|api[_-]?key|access[_-]?token|token)[[:space:]]*[:=][[:space:]]*[^[:space:]"',}]{6,}
url-with-credentials    [a-zA-Z][a-zA-Z0-9+.-]*://[^/[:space:]]+:[^/@[:space:]]+@
